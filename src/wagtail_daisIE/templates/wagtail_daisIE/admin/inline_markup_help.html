{% trans "Inline markup" %}

{% blocktrans trimmed %} Labels, help text and button labels accept a short, safe subset of Markdown. Type the markup directly — no toolbar needed. {% endblocktrans %}

  • **{% trans "bold" %}**
  • _{% trans "italic" %}_
  • __{% trans "underline" %}__
  • ~~{% trans "strikethrough" %}~~
  • [{% trans "text" %}](https://example.com) — {% trans "a link." %} {% trans "Add" %} {target=_blank} {% trans "to open it in a new tab." %}
  • [{% trans "text" %}]{color=primary} — {% blocktrans trimmed %} colour from the theme (for example primary, secondary, accent, info, success, warning, error), or use a hex value such as {color=#e11d48}. {% endblocktrans %}
  • {style="font-size: 2em"} / {.class} — {% trans "extra inline styling (safely filtered)." %}

{% trans "Double braces insert a context value, for example" %} {% verbatim %}{{ bread.name }}{% endverbatim %} {% trans "or" %} {% verbatim %}{{ user.username }}{% endverbatim %}.
