{% trans "Simple markup" %}

{% blocktrans trimmed %} Text fields accept a short, safe subset of Markdown. Type it directly — no toolbar needed. {% endblocktrans %}

  • **{% trans "bold" %}**
  • _{% trans "italic" %}_
  • __{% trans "underline" %}__
  • ~~{% trans "strikethrough" %}~~
  • [{% trans "text" %}](https://example.com)
  • [{% trans "text" %}](https://example.com){target=_blank} — {% trans "opens the link in a new tab." %}

{% trans "Colour" %}

{% blocktrans trimmed %} Wrap text in square brackets and add a colour: named values come from the active theme, or use a hex value. {% endblocktrans %}

  • [{% trans "text" %}]{color=primary} — primary, secondary, accent, neutral, info, success, warning, error {% trans "(and their content variants)." %}
  • [{% trans "text" %}]{color=#e11d48}

{% trans "Inline styling" %}

{% blocktrans trimmed %} Add a style block or class to a bracketed span. Styling is safely filtered, so only a small set of properties takes effect. {% endblocktrans %}

  • [{% trans "text" %}]{style="font-size: 2em"}
  • [{% trans "text" %}]{.badge}

{% trans "Placeholders" %}

{% trans "Double braces insert a context value, for example" %} {% verbatim %}{{ bread.name }}{% endverbatim %} {% trans "or" %} {% verbatim %}{{ user.username }}{% endverbatim %}.
